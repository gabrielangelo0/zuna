# Andares (esgoto, montanha, escadas) — plano de implementação

> **Para quem executa:** SUB-SKILL OBRIGATÓRIA: use superpowers:subagent-driven-development
> (recomendado) ou superpowers:executing-plans para implementar tarefa por tarefa. Os passos
> usam caixa (`- [ ]`) para marcar progresso.

**Objetivo:** dois estilos de terreno novos (esgoto e montanha), escada com visual próprio,
rótulo de andar na tela e escada de volta criada automaticamente pelo painel.

**Arquitetura:** andar é **mundo**, não eixo Z. A mecânica de escada já existe inteira
(`enterPortal` dispara ao pisar, trata nível mínimo, tem trava contra quicar). Este plano só
acrescenta terreno, metadado (`depth`), visual de portal (`kind`) e a criação do par de volta.

**Stack:** TypeScript (sem build step no shared — `node --experimental-strip-types`),
Drizzle ORM, Colyseus 0.18, Phaser 4, PGlite em desenvolvimento.

Spec: [docs/superpowers/specs/2026-10-03-andares-design.md](../specs/2026-10-03-andares-design.md)

## Restrições globais

- **Sem formatador.** Linhas escritas à mão, até ~130 colunas. Não rodar prettier/eslint --fix.
- **Comentários e texto de tela em português**, com acento correto. Comentário explica *por quê*,
  não *o quê*.
- **Nada de conteúdo cravado em código.** O que é conteúdo nasce de semente e o painel manda
  depois. Aqui isso vale para `depth` e `kind`: colunas com padrão, editadas no painel.
- **`npm run typecheck` tem que passar** ao fim de cada tarefa. É o único portão que o projeto
  tem hoje, além dos testes que este plano cria.
- **Testes com `node --test`**, que já vem no Node. **Nenhuma dependência nova.**
- **O campo `worlds.pvp`** nasce neste plano com padrão `"off"` e **sem tela no painel**. O
  comportamento é do spec do PvP.
- As chaves de `MONSTERS` **não batem** com os nomes mostrados: `rat` é "Slime Rosa", `troll` é
  "Yeti", `golem` é "Morcego do Musgo". Escolher pela chave põe o bicho errado.

/**
 * Tiles alcançáveis a pé desde um ponto, em quatro direções. Usado para não nascer monstro
 * preso numa ilha, e no teste que garante que um mundo novo não tranca jogador num bolsão.
 */
export function reachableTiles(map: GameMap, from: Point): Uint8Array {
  const seen = new Uint8Array(map.width * map.height);
  if (!isWalkable(map, from.x, from.y)) return seen;
  const start = from.y * map.width + from.x;
  const queue = [start];
  seen[start] = 1;
  for (let head = 0; head < queue.length; head++) {
    const index = queue[head];
    const x = index % map.width;
    const y = (index - x) / map.width;
    for (const [dx, dy] of [[1, 0], [-1, 0], [0, 1], [0, -1]]) {
      const nx = x + dx;
      const ny = y + dy;
      if (!isWalkable(map, nx, ny)) continue;
      const next = ny * map.width + nx;
      if (seen[next]) continue;
      seen[next] = 1;
      queue.push(next);
    }
  }
  return seen;
}
```

/**
 * Esgoto: canais de água em grade, passarelas de pedra nas margens, lama nos cantos sujos.
 * Os canais levam travessia (`bridge`) de propósito: uma grade de água sem travessia corta o
 * mapa em bolsões, e aí quem desce fica preso entre dois canais.
 */
function paintSewer(map: GameMap, shape: MapShape, rng: () => number) {
  const painter = { map, shape, rng };
  const { seed } = shape;
  each(map, (x, y, i) => {
    map.ground[i] = Ground.Stone;
    if (fromPlaza(shape, x, y) < shape.townRadius + 3) return;
    // A grade de canais anda com o ruído, senão o esgoto vira papel quadriculado.
    const drift = fbm(x / 16, y / 16, seed + 5) * 4;
    const canal = Math.abs(((x + drift) % 9) - 4.5) < 1.2 || Math.abs(((y + drift) % 9) - 4.5) < 1.2;
    if (canal && fbm(x / 4.5, y / 4.5, seed + 131) <= 0.58) {
      map.ground[i] = Ground.Water;
      return;
    }
    const grime = fbm(x / 6, y / 6, seed + 61);
    if (grime > 0.6) map.ground[i] = Ground.Mud;
    const r = rng();
    // Tijolo e abóbada não existem na paleta: coluna e rocha fazem o papel de parede.
    if (grime > 0.72 && r < 0.3) map.objects[i] = r < 0.18 ? Obj.Pillar : Obj.Rock;
    else if (r < 0.03) map.objects[i] = Obj.Mushroom;
    else if (r < 0.042) map.objects[i] = Obj.Barrel;
  });
  clearings(painter, Math.round((map.width * map.height) / 700), Ground.Stone);
  carveRoads(painter, Ground.Stone);
  fence(map, Ground.Stone, Obj.Rock);
  clearPlaza(map, shape);
}
```

```ts
/**
 * Montanha: a altitude sobe conforme afasta da praça, então terra vira pedra e pedra vira
 * neve. Os penhascos (`cliff`) são linhas finas de rocha de propósito: paredão largo fecharia
 * platô e deixaria jogador sem rota.
 */
function paintMountain(map: GameMap, shape: MapShape, rng: () => number) {
  const painter = { map, shape, rng };
  const { seed } = shape;
  const alcance = Math.max(map.width, map.height) * 0.42;
  each(map, (x, y, i) => {
    const subida = Math.min(1, fromPlaza(shape, x, y) / alcance);
    const altura = subida * 0.75 + fbm(x / 13, y / 13, seed + 29) * 0.45;
    map.ground[i] = altura > 0.72 ? Ground.Snow : altura > 0.38 ? Ground.Stone : Ground.Dirt;
    if (fromPlaza(shape, x, y) < shape.townRadius + 3) return;
    if (Math.abs(fbm(x / 9, y / 9, seed + 53) - 0.5) < 0.055) {
      map.objects[i] = Obj.Rock;
      return;
    }
    const r = rng();
    if (map.ground[i] === Ground.Snow) {
      if (r < 0.1) map.objects[i] = Obj.SnowPine;
      else if (r < 0.125) map.objects[i] = Obj.Crystal;
      return;
    }
    if (map.ground[i] === Ground.Stone) {
      if (r < 0.08) map.objects[i] = Obj.Rock;
      else if (r < 0.095) map.objects[i] = Obj.Crystal;
      return;
    }
    if (r < 0.16) map.objects[i] = Obj.Pine;
    else if (r < 0.19) map.objects[i] = Obj.Fern;
  });
  clearings(painter, Math.round((map.width * map.height) / 800), null);
  carveRoads(painter, Ground.Dirt);
  fence(map, Ground.Stone, Obj.Rock);
  clearPlaza(map, shape);
}
```

  stairsUp: "Escada para cima",
  ladderDown: "Rampa para baixo",
  ladderUp: "Rampa para cima",
  hole: "Buraco (mão única)",
};

/** O sentido contrário, para a escada de volta. O buraco volta como escada: por buraco não se sobe. */
export function returnKind(kind: PortalKind): PortalKind {
  switch (kind) {
    case "stairsDown":
      return "stairsUp";
    case "stairsUp":
      return "stairsDown";
    case "ladderDown":
      return "ladderUp";
    case "ladderUp":
      return "ladderDown";
    case "hole":
      return "stairsUp";
    default:
      return "portal";
  }
}

export function clampDepth(value: unknown): number {
  const depth = Math.round(Number(value));
  if (!Number.isFinite(depth)) return 0;
  return Math.min(WORLD_MAX_DEPTH, Math.max(WORLD_MIN_DEPTH, depth));
}

/** "Esgoto de Zuna · −1". Andar 0 é a superfície e não leva sufixo. */
export function floorLabel(name: string, depth: number): string {
  if (depth === 0) return name;
  // Sinal de menos de verdade (U+2212), não hífen: alinha com os números na fonte da HUD.
  return `${name} · ${depth > 0 ? "+" : "−"}${Math.abs(depth)}`;
}
